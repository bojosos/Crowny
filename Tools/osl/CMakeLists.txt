cmake_minimum_required(VERSION 3.24)
project(CrownyOslCompiler LANGUAGES C CXX)
find_package(OSL CONFIG REQUIRED)
find_package(LLVM CONFIG REQUIRED)
if(NOT LLVM_VERSION_MAJOR EQUAL 20)
    message(FATAL_ERROR "This experimental adapter is tested with LLVM 20 only")
endif()
add_executable(crowny-osl Compiler.cpp)
target_compile_features(crowny-osl PRIVATE cxx_std_17)
target_include_directories(crowny-osl SYSTEM PRIVATE ${LLVM_INCLUDE_DIRS})
target_compile_definitions(crowny-osl PRIVATE ${LLVM_DEFINITIONS})
llvm_map_components_to_libnames(llvm_libs Core IRReader BitReader BitWriter Passes Linker native)
target_link_libraries(crowny-osl PRIVATE OSL::oslexec OSL::oslquery ${llvm_libs})
add_executable(crowny-osl-reference Reference.cpp)
target_compile_features(crowny-osl-reference PRIVATE cxx_std_17)
target_link_libraries(crowny-osl-reference PRIVATE OSL::oslexec OSL::oslquery)
